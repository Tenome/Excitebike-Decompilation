# Native Excitebike: the lifted C game + picture/sound layers + SDL2 front end. Build from the repository root:
#   cmake -S port -B build/port -DCMAKE_PREFIX_PATH=<folder with SDL2 (sdl2-config.cmake)>   (clang or gcc recommended)
#   cmake --build build/port
# Run:  excitebike orig/excitebike_us.nes     (keys: arrows, Z = A/accelerate, X = B/turbo, Enter = Start, Shift = Select, F11 fullscreen, Esc quit)
cmake_minimum_required(VERSION 3.16)
project(excitebike C)
set(CMAKE_C_STANDARD 11)
find_package(SDL2 REQUIRED)
add_executable(excitebike main.c ppu.c apu.c audiomod.c ../c/excitebike.c)
target_include_directories(excitebike PRIVATE ../c ${SDL2_INCLUDE_DIRS})
target_compile_options(excitebike PRIVATE $<$<NOT:$<C_COMPILER_ID:MSVC>>:-Wno-unused-label -Wno-unused-function>)
if(TARGET SDL2::SDL2)
  target_link_libraries(excitebike PRIVATE SDL2::SDL2)
else()
  target_link_libraries(excitebike PRIVATE ${SDL2_LIBRARIES})
endif()
if(UNIX)
  target_link_libraries(excitebike PRIVATE m)
endif()
